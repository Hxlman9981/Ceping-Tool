  startQuiz() {

    const s = this.state;

      const el = document.getElementById('timerDisplay');

      if(el) el.textContent = this.formatTime(s.elapsed);

    }, 1000);

    this.render();

  },

 

  formatTime(sec) {

    const m = Math.floor(sec / 60).toString().padStart(2,'0');

    const s = (sec % 60).toString().padStart(2,'0');

    return m + ':' + s;

  selectOption(idx) {

    if(this.state.answers[this.state.currentQ] !== undefined) return;

 

    if(this.state.autoNextTimer) clearTimeout(this.state.autoNextTimer);

 

    const q = this.state.questions[this.state.currentQ];

    const isCorrect = idx === q.answer;

    this.state.answers[this.state.currentQ] = idx;

    if(isCorrect) this.state.score++;

 

    this.render();

 

    if(this.state.currentQ < this.state.questions.length - 1) {

      this.state.autoNextTimer = setTimeout(() => {

        this.nextQuestion();

      }, 3000);

    }

  prevQuestion() {

    if(this.state.autoNextTimer) clearTimeout(this.state.autoNextTimer);

    this.state.autoNextTimer = null;

 

    if(this.state.currentQ > 0) {

      this.state.currentQ--;

      this.render();

    }

  nextQuestion() {

    if(this.state.autoNextTimer) clearTimeout(this.state.autoNextTimer);

    this.state.autoNextTimer = null;

 

    if(this.state.answers[this.state.currentQ] === undefined) return;

 

    if(this.state.currentQ < this.state.questions.length - 1) {

      this.state.currentQ++;

      this.render();

    } else {

      this.finishQuiz();

    }
